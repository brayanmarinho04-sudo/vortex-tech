-- Data prevista de entrega ALTER TABLE public.vortex_briefings ADD COLUMN IF NOT EXISTS data_prevista date; -- RLS continua ativada ALTER TABLE public.vortex_briefings ENABLE ROW LEVEL SECURITY; -- Acesso autenticado para consultar briefings: -- somente usuários cujo perfil tenha cargo = admin. GRANT SELECT, UPDATE ON public.vortex_briefings TO authenticated; DROP POLICY IF EXISTS "Admins podem consultar briefings" ON public.vortex_briefings; CREATE POLICY "Admins podem consultar briefings" ON public.vortex_briefings FOR SELECT TO authenticated USING ( EXISTS ( SELECT 1 FROM public.profiles p WHERE p.id = (SELECT auth.uid()) AND p.cargo = 'admin' ) ); DROP POLICY IF EXISTS "Admins podem atualizar briefings" ON public.vortex_briefings; CREATE POLICY "Admins podem atualizar briefings" ON public.vortex_briefings FOR UPDATE TO authenticated USING ( EXISTS ( SELECT 1 FROM public.profiles p WHERE p.id = (SELECT auth.uid()) AND p.cargo = 'admin' ) ) WITH CHECK ( EXISTS ( SELECT 1 FROM public.profiles p WHERE p.id = (SELECT auth.uid()) AND p.cargo = 'admin' ) );
